"""只供本轮独立诊断/验收；不改业务源码，不保存密钥。"""
from decimal import Decimal
from pathlib import Path
from datetime import datetime, timezone
import json
import os
import sys
from urllib import request

ROOT=Path(__file__).resolve().parents[2]
if str(ROOT) not in sys.path:sys.path.append(str(ROOT))
from agent.model_client import ModelClient, ModelError, _NoRedirect, load_config

LEDGER=Path(__file__).with_name('api-budget.json')
CAP=Decimal('5.00');RESERVATION=Decimal('0.25')


def save(value):
    temporary=LEDGER.with_suffix('.tmp')
    temporary.write_text(json.dumps(value,ensure_ascii=False,indent=2),encoding='utf-8')
    temporary.replace(LEDGER)


class BoundedOpener:
    def __init__(self,name):self.name=name;self.opener=request.build_opener(_NoRedirect())
    def open(self,req,timeout):
        body=json.loads(req.data)
        if req.full_url!='https://api.deepseek.com/chat/completions' or body.get('model')!='deepseek-flash':
            raise ModelError('本轮只授权当前DeepSeek Flash官方目标，未发送。')
        if len(req.data)>65536 or body.get('max_tokens',0)>4096 or body.get('thinking')!={'type':'disabled'}:
            raise ModelError('请求超过费用预留边界，未发送。')
        value=json.loads(LEDGER.read_text(encoding='utf-8')) if LEDGER.exists() else dict(
            limit_CNY='5.00',reserve_each_CNY='0.25',authorization='2026-10-26 用户授权实际银行API测试不超过5元',
            pricing_source='https://api-docs.deepseek.com/zh-cn/quick_start/pricing/',
            peak_input_miss_per_million_CNY='2.00',peak_output_per_million_CNY='8.00',calls=[])
        if (len(value['calls'])+1)*RESERVATION>CAP:raise ModelError('费用预留已达5元上限，未发送。')
        value['calls'].append(dict(name=self.name,status='reserved_unknown',reserved_CNY='0.25',
            at_utc=datetime.now(timezone.utc).isoformat(),request_bytes=len(req.data)))
        save(value)
        return self.opener.open(req,timeout=timeout)


def call(messages,name,config=None):
    lock=LEDGER.with_suffix('.lock');fd=os.open(lock,os.O_CREAT|os.O_EXCL|os.O_WRONLY)
    try:
        config=config or load_config()
        if config.provider!='deepseek' or config.model!='deepseek-flash':raise ModelError('当前配置不是已定价的DeepSeek Flash，停止测试。')
        client=ModelClient(config,opener=BoundedOpener(name))
        text=client(messages)
        value=json.loads(LEDGER.read_text(encoding='utf-8'))
        usage=client.last_metadata.get('usage',{})
        estimate=(Decimal(usage.get('prompt_tokens',0))*2+Decimal(usage.get('completion_tokens',0))*8)/1000000
        value['calls'][-1].update(status='response_received',usage=usage,metadata=client.last_metadata,
            conservative_usage_CNY=str(estimate) if {'prompt_tokens','completion_tokens'}<=usage.keys() else None)
        save(value)
        return text,client.last_metadata
    finally:
        os.close(fd);lock.unlink()


if __name__=='__main__':
    text,metadata=call([dict(role='system',content='只返回JSON对象，查余额意图为balance，不执行任何业务。'),
        dict(role='user',content='帮我看看还有多少钱可用')],'connectivity')
    passed=json.loads(text).get('intent')=='balance'
    result=dict(passed=passed,actual_http=True,metadata=metadata,model_key_saved=False,
        note='高峰、全部输入未命中计价为用量估算上界，不是账户实际扣费；每请求先预留0.25元，最多20次。')
    Path(__file__).with_name('api-connectivity.json').write_text(json.dumps(result,ensure_ascii=False,indent=2),encoding='utf-8')
    print(json.dumps(result,ensure_ascii=False))
    raise SystemExit(0 if passed else 1)
